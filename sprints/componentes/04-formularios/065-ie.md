# FiscalUI Framework

## Documento 065 — IE

**Nível 4 — Formulários**

**Versão 1.0**

Campo de Inscrição Estadual com máscara por estado. A validação específica de cada UF é delegada ao FiscalCore.

---

```js
class UIInscricaoEstadual extends UIComponent {
    constructor(options = {}) {
        super(options);
        this.name = options.name || '';
        this.label = options.label || 'Inscrição Estadual';
        this.value = options.value || '';
        this.uf = options.uf || '';
        this.placeholder = options.placeholder || '000.000.000.000';
        this.rules = options.rules || [];
    }

    static MASKS = {
        SP: '000.000.000.000',
        RJ: '00.000.00-0',
        MG: '000.000.000/0000',
        RS: '000/0000000',
        PR: '000.00000-00',
        SC: '000.000.000',
        BA: '000000-00',
        PE: '0000000-00',
        CE: '00000000-0',
        default: '000.000.000.000'
    };

    template() {
        return `
            <div class="ui-field">
                ${this.label ? `<label class="ui-field__label">${this.label}</label>` : ''}
                <div class="ui-field__input-wrapper">
                    <input class="ui-field__input ui-field__input--ie" type="text" inputmode="numeric"
                           name="${this.name}" placeholder="${this.placeholder}"
                           value="${this.value || ''}" autocomplete="off">
                </div>
                <span class="ui-field__error"></span>
            </div>
        `;
    }

    onInit() {
        this._input = this.query('.ui-field__input');

        this._input.addEventListener('input', () => {
            const raw = this._input.value.replace(/\D/g, '');
            const maxLen = (UIInscricaoEstadual.MASKS[this.uf] || UIInscricaoEstadual.MASKS.default).replace(/\D/g, '').length;
            this._input.value = this._mask(raw.slice(0, maxLen));
            this.value = raw;
            this.emit('field:change', { name: this.name, value: this.value });
        });

        this._input.addEventListener('blur', () => this.emit('field:blur', { name: this.name }));
    }

    _mask(raw) {
        const mask = UIInscricaoEstadual.MASKS[this.uf] || UIInscricaoEstadual.MASKS.default;
        let result = '';
        let idx = 0;
        for (const char of mask) {
            if (idx >= raw.length) break;
            if (char === '0') { result += raw[idx]; idx++; }
            else { result += char; }
        }
        return result;
    }

    value() { return this.value; }
    setValue(val) { this.value = val.replace(/\D/g, ''); if (this._input) this._input.value = this._mask(this.value); }
    reset() { this.setValue(''); }
    setUF(uf) { this.uf = uf; if (this._input) this._input.value = this._mask(this.value); }
}
```
