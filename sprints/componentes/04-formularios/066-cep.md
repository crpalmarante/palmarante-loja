# FiscalUI Framework

## Documento 066 — CEP

**Nível 4 — Formulários**

**Versão 1.0**

Campo de CEP com máscara (#####-###), busca automática por API de CEP e preenchimento de endereço.

---

```js
class UICEPSearch extends UIComponent {
    constructor(options = {}) {
        super(options);
        this.name = options.name || '';
        this.label = options.label || 'CEP';
        this.value = options.value || '';
        this.placeholder = options.placeholder || '00000-000';
        this.autoSearch = options.autoSearch !== false;
        this.searchEndpoint = options.searchEndpoint || 'https://viacep.com.br/ws/{cep}/json/';
        this.onAddressFound = options.onAddressFound || null;
        this.rules = options.rules || [];
    }

    template() {
        return `
            <div class="ui-field">
                ${this.label ? `<label class="ui-field__label">${this.label}</label>` : ''}
                <div class="ui-field__input-wrapper">
                    <span class="ui-field__prefix-icon">${FiscalUI.icons.render('map-pin', { size: 16 })}</span>
                    <input class="ui-field__input ui-field__input--cep" type="text" inputmode="numeric"
                           name="${this.name}" placeholder="${this.placeholder}"
                           value="${this.value || ''}" maxlength="9" autocomplete="postal-code">
                    <span class="ui-field__cep-loading" hidden>${FiscalUI.icons.render('loader', { size: 16 })}</span>
                </div>
                <span class="ui-field__error"></span>
            </div>
        `;
    }

    onInit() {
        this._input = this.query('.ui-field__input');
        this._loading = this.query('.ui-field__cep-loading');
        this._timer = null;

        this._input.addEventListener('input', () => {
            const raw = this._input.value.replace(/\D/g, '').slice(0, 8);
            this._input.value = this._mask(raw);
            this.value = raw;

            clearTimeout(this._timer);
            if (raw.length === 8 && this.autoSearch) {
                this._timer = setTimeout(() => this._search(raw), 500);
            }
            this.emit('field:change', { name: this.name, value: this.value });
        });

        this._input.addEventListener('blur', () => {
            if (this.value.length === 8) this._search(this.value);
            this.emit('field:blur', { name: this.name });
        });
    }

    _mask(raw) {
        return raw.replace(/^(\d{5})(\d)/, '$1-$2');
    }

    async _search(cep) {
        this._loading.hidden = false;
        try {
            const url = this.searchEndpoint.replace('{cep}', cep);
            const response = await fetch(url);
            if (!response.ok) throw new Error('CEP não encontrado');
            const data = await response.json();
            if (data.erro) throw new Error('CEP não encontrado');
            this.onAddressFound?.(data);
            this.emit('cep:found', { cep, address: data });
            this.query('.ui-field__error').textContent = '';
            this.element.classList.remove('ui-field--error');
        } catch (err) {
            this.element.classList.add('ui-field--error');
            this.query('.ui-field__error').textContent = err.message || 'CEP não encontrado';
        } finally {
            this._loading.hidden = true;
        }
    }

    value() { return this.value; }
    setValue(val) { this.value = val.replace(/\D/g, ''); if (this._input) this._input.value = this._mask(this.value); }
    reset() { this.setValue(''); this.element.classList.remove('ui-field--error'); this.query('.ui-field__error').textContent = ''; }
}
```
