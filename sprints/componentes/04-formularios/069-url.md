# FiscalUI Framework

## Documento 069 — URL

**Nível 4 — Formulários**

**Versão 1.0**

Campo de URL com preview de link, validação de formato e atalho para abrir em nova aba.

---

```js
class UIURL extends UIComponent {
    constructor(options = {}) {
        super(options);
        this.name = options.name || '';
        this.label = options.label || 'URL';
        this.value = options.value || '';
        this.placeholder = options.placeholder || 'https://exemplo.com.br';
        this.showPreview = options.showPreview !== false;
        this.rules = options.rules || [{ type: 'url' }];
    }

    template() {
        return `
            <div class="ui-field">
                ${this.label ? `<label class="ui-field__label">${this.label}</label>` : ''}
                <div class="ui-field__input-wrapper">
                    <span class="ui-field__prefix-icon">${FiscalUI.icons.render('link', { size: 16 })}</span>
                    <input class="ui-field__input ui-field__input--url" type="url" inputmode="url"
                           name="${this.name}" placeholder="${this.placeholder}"
                           value="${this.value || ''}" autocomplete="url">
                </div>
                <span class="ui-field__error"></span>
            </div>
        `;
    }

    onInit() {
        this._input = this.query('.ui-field__input');
        this._error = this.query('.ui-field__error');

        this._input.addEventListener('input', () => {
            this.value = this._input.value;
            if (this.showPreview && this._validate(this.value)) this._showPreview();
            else this._removePreview();
            this.emit('field:change', { name: this.name, value: this.value });
        });

        this._input.addEventListener('blur', () => {
            if (this.value && !this._validate(this.value)) {
                this.element.classList.add('ui-field--error');
                this._error.textContent = 'URL inválida';
            }
            this.emit('field:blur', { name: this.name });
        });
    }

    _validate(url) {
        try { new URL(url); return true; }
        catch { return /^https?:\/\/.+\..+/.test(url); }
    }

    _showPreview() {
        this._removePreview();
        const preview = document.createElement('a');
        preview.className = 'ui-url__preview';
        preview.href = this.value;
        preview.target = '_blank';
        preview.rel = 'noopener noreferrer';
        preview.textContent = this.value.length > 50 ? this.value.slice(0, 47) + '...' : this.value;
        preview.innerHTML += ` ${FiscalUI.icons.render('external-link', { size: 12 })}`;
        this.element.appendChild(preview);
    }

    _removePreview() { this.element.querySelector('.ui-url__preview')?.remove(); }

    value() { return this.value; }
    setValue(val) { this.value = val; if (this._input) this._input.value = val; }
    reset() { this.setValue(''); this._removePreview(); this.element.classList.remove('ui-field--error'); }
}
```
