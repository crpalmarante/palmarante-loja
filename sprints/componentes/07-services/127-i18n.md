# FiscalUI Framework

## Documento 127 — i18n

**Nível 7 — Services**

**Versão 1.0**

Serviço de internacionalização com carregamento lazy de dicionários, pluralização e interpolação.

---

```js
class I18nService {
    constructor(options = {}) {
        this.locale = options.locale || navigator.language?.split('-')[0] || 'pt-BR';
        this.fallbackLocale = options.fallbackLocale || 'pt-BR';
        this.dictionaries = {};
        this._loaded = {};
        this._listeners = [];
    }

    async loadLocale(locale, url) {
        if (this._loaded[locale]) return;
        const res = await fetch(url);
        this.dictionaries[locale] = await res.json();
        this._loaded[locale] = true;
        if (locale === this.locale) this._notify();
    }

    async setLocale(locale) {
        if (locale === this.locale) return;
        this.locale = locale;
        if (!this.dictionaries[locale]) {
            console.warn(`[i18n] Locale '${locale}' não carregado. Usando '${this.fallbackLocale}'.`);
        }
        this._notify();
    }

    t(key, params = {}) {
        const dict = this.dictionaries[this.locale] || this.dictionaries[this.fallbackLocale] || {};
        let value = this._resolveKey(dict, key);

        if (!value) {
            console.warn(`[i18n] Chave '${key}' não encontrada`);
            return key;
        }

        if (params.count !== undefined) {
            value = this._pluralize(value, params.count);
        }

        return value.replace(/\{\{(\w+)\}\}/g, (_, k) => params[k] !== undefined ? params[k] : `{{${k}}}`);
    }

    _resolveKey(obj, key) {
        return key.split('.').reduce((acc, part) => acc?.[part], obj);
    }

    _pluralize(value, count) {
        if (typeof value === 'object') {
            const pluralKey = count === 1 ? 'one' : 'other';
            return value[pluralKey] || value.other || '';
        }
        return value;
    }

    onChange(cb) { this._listeners.push(cb); }

    _notify() { this._listeners.forEach(cb => cb(this.locale)); }
}
```
